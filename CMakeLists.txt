cmake_minimum_required(VERSION 3.21)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" JELLI_VERSION LIMIT_COUNT 1)
project(jelligotchi VERSION ${JELLI_VERSION} LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(JELLI_BUILD_SDL "Build the SDL2 desktop emulator" ON)
option(JELLI_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(
    -Wall -Wextra -Wpedantic -Werror
    -Wconversion -Wsign-conversion -Wshadow
    -Wstrict-prototypes -Wmissing-prototypes
    -Wformat=2 -Wundef -Wcast-align -Wwrite-strings
    -Wvla -Wdouble-promotion -Wswitch-enum
  )
  if(JELLI_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
elseif(MSVC)
  add_compile_options(/W4 /WX)
endif()
include(cmake/JelliCollection.cmake)
jelli_collection_data(collection_data)
include(cmake/JelliFood.cmake)
jelli_food_data(food_data)
include(cmake/JelliExercise.cmake)
jelli_exercise_data(exercise_data)
include(cmake/JelliWake.cmake)
jelli_wake_data(wake_data)
include(cmake/JelliActivities.cmake)
jelli_activities_data(activities_data)
include(cmake/JelliPotty.cmake)
jelli_potty_data(potty_data)
include(cmake/JelliBehaviors.cmake)
jelli_behaviors_data(behaviors_data)
add_library(jelli_core STATIC ${wake_data} core/wake.c core/creature.c ${activities_data} core/activities.c ${potty_data} core/potty.c ${behaviors_data} core/behavior.c core/save_tail.c ${exercise_data} ${food_data} ${collection_data} core/collection_migrate.c core/collection.c core/nutrition.c core/engine.c core/gesture.c core/render.c core/animation.c
  core/prizes.c core/habits.c core/sleep_log.c core/tunables.c core/sound.c core/particles.c core/game.c core/events.c core/personality.c core/game_events.c core/game_api.c core/game_commands.c core/game_time.c core/game_advance.c core/save.c)
target_include_directories(jelli_core PUBLIC include)
add_executable(jelli_sound_export tools/audio/export.c)
target_link_libraries(jelli_sound_export PRIVATE jelli_core)
# The pet UI library also builds without SDL for Jelli Art's game preview.
option(JELLI_BUILD_PET "Build the pet UI library and game preview without SDL" OFF)
if(JELLI_BUILD_SDL OR JELLI_BUILD_PET)
  include(cmake/JelliPetAssets.cmake)
  jelli_generate_pet_assets(pet_assets)
  add_library(jelli_pet STATIC core/debug.c core/debug_clock.c core/debug_habits.c core/debug_cheats.c core/debug_events.c core/debug_sound.c core/debug_tunables.c core/debug_state.c core/pet_engine.c core/pet_rewards.c core/pet_bubbles.c core/pet_atmosphere.c core/pet_timing.c core/pet_actions.c core/pet_clock.c core/pet_layout.c core/pet_actor.c core/pet_health.c core/pet_feedback.c core/pet_ui.c core/pet_particles.c core/pet_food.c core/pet_collection.c core/pet_gallery.c core/asset_source.c core/pet_canvas.c core/pet_menu.c core/pet_behavior_draw.c core/pet_draw.c core/pet_render.c ${pet_assets})
  target_compile_definitions(jelli_pet PRIVATE JELLI_VERSION_LABEL="v${JELLI_VERSION}")
  target_link_libraries(jelli_pet PUBLIC jelli_core)
  add_executable(jelli_game_preview tools/game-preview/preview.c)
  target_link_libraries(jelli_game_preview PRIVATE jelli_pet)
endif()
if(JELLI_BUILD_SDL)
  find_package(SDL2 2.0.18 REQUIRED CONFIG)
  if(WIN32)
    set(jelli_debug_transport ports/sdl/debug_socket_win.c)
  else()
    set(jelli_debug_transport ports/sdl/debug_socket.c)
  endif()
  add_executable(jelligotchi ports/sdl/main.c ports/sdl/asset_reload.c ports/sdl/asset_pack.c ports/sdl/options.c ports/sdl/session.c
    ports/sdl/storage.c ports/sdl/sound_output.c ports/sdl/demo.c ${jelli_debug_transport})
  target_link_libraries(jelligotchi PRIVATE jelli_pet SDL2::SDL2)
  if(WIN32)
    add_custom_command(TARGET jelligotchi POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2>
              $<TARGET_FILE_DIR:jelligotchi> VERBATIM)
  endif()
endif()
