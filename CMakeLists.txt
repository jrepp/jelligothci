cmake_minimum_required(VERSION 3.21)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" JELLI_VERSION LIMIT_COUNT 1)
project(jelligotchi VERSION ${JELLI_VERSION} LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(JELLI_BUILD_SDL "Build the SDL2 desktop emulator" ON)
option(JELLI_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(
    -Wall -Wextra -Wpedantic -Werror
    -Wconversion -Wsign-conversion -Wshadow
    -Wstrict-prototypes -Wmissing-prototypes
    -Wformat=2 -Wundef -Wcast-align -Wwrite-strings
    -Wvla -Wdouble-promotion -Wswitch-enum
  )
  if(JELLI_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
elseif(MSVC)
  add_compile_options(/W4 /WX)
endif()
add_library(jelli_core STATIC core/engine.c core/render.c core/animation.c
  core/game.c core/game_commands.c core/game_time.c core/game_advance.c core/save.c)
target_include_directories(jelli_core PUBLIC include)
if(JELLI_BUILD_SDL)
  find_package(SDL2 2.0.18 REQUIRED CONFIG)
  include(cmake/JelliPetAssets.cmake)
  jelli_generate_pet_assets(pet_assets)
  add_library(jelli_pet STATIC core/pet_engine.c core/pet_ui.c core/pet_render.c ${pet_assets})
  target_link_libraries(jelli_pet PUBLIC jelli_core)
  add_executable(jelligotchi ports/sdl/main.c ports/sdl/options.c ports/sdl/session.c
    ports/sdl/storage.c ports/sdl/demo.c)
  target_link_libraries(jelligotchi PRIVATE jelli_pet SDL2::SDL2)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(jelli_tests tests/test_engine.c)
  target_link_libraries(jelli_tests PRIVATE jelli_core)
  add_test(NAME engine COMMAND jelli_tests)
  add_executable(jelli_animation_tests tests/test_animation.c)
  target_link_libraries(jelli_animation_tests PRIVATE jelli_core)
  add_test(NAME animation COMMAND jelli_animation_tests)
  add_executable(jelli_render_tests tests/test_render.c)
  target_link_libraries(jelli_render_tests PRIVATE jelli_core)
  add_test(NAME render COMMAND jelli_render_tests)
  foreach(component game save)
    add_executable(jelli_${component}_tests tests/test_${component}.c)
    target_link_libraries(jelli_${component}_tests PRIVATE jelli_core)
    add_test(NAME ${component} COMMAND jelli_${component}_tests)
  endforeach()
  add_executable(jelli_storage_tests tests/test_save_storage.c ports/sdl/storage.c)
  target_include_directories(jelli_storage_tests PRIVATE ports/sdl)
  target_link_libraries(jelli_storage_tests PRIVATE jelli_core)
  add_test(NAME save_storage COMMAND jelli_storage_tests)
  if(JELLI_BUILD_SDL)
    add_executable(jelli_pet_engine_tests tests/test_pet_engine.c)
    target_link_libraries(jelli_pet_engine_tests PRIVATE jelli_pet)
    add_test(NAME pet_engine COMMAND jelli_pet_engine_tests)
    add_executable(jelli_pet_render_tests tests/test_pet_render.c)
    target_link_libraries(jelli_pet_render_tests PRIVATE jelli_pet)
    add_test(NAME pet_render COMMAND jelli_pet_render_tests)
    add_test(NAME sdl_shapes_smoke COMMAND jelligotchi --shapes --headless --frames 5)
    add_test(NAME sdl_pet_smoke COMMAND jelligotchi --pet --headless --frames 5)
    add_test(NAME sdl_pet_demo COMMAND jelligotchi --pet --headless --demo --frames 780)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME pet_session COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pet_session.py $<TARGET_FILE:jelligotchi>)
    endif()
  endif()
endif()
