cmake_minimum_required(VERSION 3.21)
project(jelligotchi VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(JELLI_BUILD_SDL "Build the SDL2 desktop emulator" ON)
option(JELLI_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  if(JELLI_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(jelli_core STATIC core/engine.c core/render.c)
target_include_directories(jelli_core PUBLIC include)
if(JELLI_BUILD_SDL)
  find_package(SDL2 2.0.18 REQUIRED CONFIG)
  add_executable(jelligotchi ports/sdl/main.c)
  target_link_libraries(jelligotchi PRIVATE jelli_core SDL2::SDL2)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(jelli_tests tests/test_engine.c)
  target_link_libraries(jelli_tests PRIVATE jelli_core)
  add_test(NAME engine COMMAND jelli_tests)
  if(JELLI_BUILD_SDL)
    add_test(NAME sdl_smoke COMMAND jelligotchi --headless --frames 5)
  endif()
endif()
